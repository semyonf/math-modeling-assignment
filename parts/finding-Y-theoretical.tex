\part{НАХОЖДЕНИЕ $y^T$}
    \section{Составление дифференциального уравнения}
        Передаточная функция объекта представляется в виде отношения двух многочленов:
        $$
            W(s) = \frac{k(1-as)}{(1+as)(1+b_1s+b_2s^2)} = \frac{P(s)}{Q(s)},
        $$
        $$
            P(s) = k(1-as), \qquad
            Q(s) = 1 + (a+b_1)s + (b_2+ab_1)s^2 + ab_2s^3.
        $$

        Введём промежуточную переменную $u$ так, чтобы
        $$
            \begin{cases}
                \dfrac{u(s)}{x(s)} = \dfrac{1}{Q(s)}
                \\\\
                \dfrac{y(s)}{u(s)} = P(s)
            \end{cases}
            \Rightarrow
            \begin{cases}
                x(s) = Q(s) \cdot u(s)
                \\
                y(s) = P(s) \cdot u(s)
            \end{cases}
        $$

        Переходя к оригиналам при нулевых начальных условиях ($s^n \to \dfrac{d^n}{dt^n}$):
        $$
            \begin{aligned}
                x(t) &= u(t) + (a+b_1)\dot{u}(t) + (b_2+ab_1)\ddot{u}(t) + ab_2\dddot{u}(t),
                \\
                y(t) &= k\,u(t) - ka\,\dot{u}(t).
            \end{aligned}
        $$

        Разрешим первое уравнение относительно старшей производной:
        $$
            \dddot{u}(t) = \frac{1}{ab_2} \cdot \underbrace{\Big(x(t) - u(t) - (a+b_1)\dot{u}(t) - (b_2 + ab_1)\ddot{u}(t)\Big)}_{\Omega}.
        $$

    \section{Переход к системе уравнений первого порядка}
        $$
            \begin{matrix}
                \begin{aligned}
                    z_1 &= u
                    \\
                    z_2 &= \dot{u} = \dot{z}_1
                    \\
                    z_3 &= \ddot{u} = \dot{z}_2
                \end{aligned}
                &
                \Rightarrow
                &
                \begin{cases}
                    \dot{z}_1 = z_2
                    \\
                    \dot{z}_2 = z_3
                    \\
                    \dot{z}_3 = \dfrac{x - z_1 - (a+b_1)z_2 - (b_2+ab_1)z_3}{ab_2} = \dfrac{\Omega}{ab_2}
                \end{cases}
                &
                y = k(z_1 - az_2)
            \end{matrix}
        $$

    \section{Решение системы методом Эйлера}
        Каждая переменная состояния получает приращение, равное шагу $h$, умноженному на её производную, вычисленную в предыдущей точке:
        $$
            \begin{cases}
                z_{1,n+1} = z_{1,n} + h \cdot z_{2,n}
                \\
                z_{2,n+1} = z_{2,n} + h \cdot z_{3,n}
                \\
                z_{3,n+1} = z_{3,n} + h \cdot \dfrac{\Omega_n}{ab_2}
            \end{cases}
            \qquad
            y_n = k(z_{1,n} - az_{2,n}),
        $$
        где $\Omega_n = x - z_{1,n} - (a+b_1)z_{2,n} - (b_2+ab_1)z_{3,n}$, начальные условия нулевые.

        Шаг интегрирования $h = \StepH$, выход наблюдается с интервалом $\Delta t = 1$ (каждые $100$ шагов), всего $25$ наблюдений при $t = 1, \ldots, 25$.

    \section{Проверка}
        \subsection*{Устойчивость и установившееся значение}
            Полюса передаточной функции -- корни $Q(s)$:
            $$
                s_1 = -\frac{1}{a} = -\frac{1}{3}, \qquad
                s_{2,3} = \frac{-b_1 \pm \sqrt{b_1^2 - 4b_2}}{2b_2} = -0.25 \pm 0.661i.
            $$
            Все полюса лежат в левой полуплоскости, объект устойчив, поэтому применима теорема о конечном значении. При $x(t) = C = 5$, $x(s) = C/s$:
            $$
                \lim_{t\to\infty} y(t) = \lim_{s\to0} s \cdot W(s) \cdot \frac{C}{s} = C \cdot W(0) = C \cdot k = 5 \cdot 4 = 20.
            $$
            Последнее вычисленное значение $y^T(25) = \YTheorLast$.

        \subsection*{Сравнение с точным решением}
            Совпадение установившегося значения -- необходимое, но не достаточное условие правильности: оно не зависит от коэффициентов при производных и не проверяет переходный процесс. Поэтому решение сравнивается с точной переходной характеристикой. Так как полюса $s_j$ простые, по теореме о вычетах
            $$
                y(t) = C\left(W(0) + \sum_{j=1}^{3} \frac{P(s_j)}{s_j\,Q'(s_j)}\,e^{s_jt}\right),
                \qquad Q'(s) = 3ab_2s^2 + 2(b_2+ab_1)s + (a+b_1).
            $$
            Максимальное отклонение решения методом Эйлера от точного в точках наблюдения составляет $\EulerMaxDev$, то есть менее $0.5\%$ от установившегося значения. Шаг $h = \StepH$ достаточен.

    \section{Выводы первой части}
        \begin{itemize}
            \item Числитель $W(s)$ имеет нуль в правой полуплоскости ($s = 1/a$), то есть объект неминимально"=фазовый. Поэтому в начале переходного процесса выход отклоняется в сторону, противоположную установившемуся значению (минимум около $-7.7$ при $t \approx 2.5$).
            \item Пара комплексных полюсов даёт затухающие колебания: перерегулирование до $\approx 20.7$ при $t \approx 9$, затем небольшой провал и выход на уровень $kx = 20$.
            \item Затухание определяется полюсами $-1/3$ и $-0.25 \pm 0.661i$: огибающая $e^{-0.25t}$ к $t = 25$ уменьшается в $\approx 500$ раз, в трубку $\pm2\%$ процесс входит примерно при $t \approx 15.5$. Поэтому интервал наблюдения $t \in [1; 25]$ из $25$ точек покрывает весь переходный процесс.
            \item Решение методом Эйлера совпадает с точным решением с погрешностью не более $\EulerMaxDev$.
        \end{itemize}
